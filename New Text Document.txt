// ========== ROUTING ==========
function navigate(page, projectId) {
  state.currentPage = page;
  state.currentProject = projectId || null;
  // Update nav items
  document.querySelectorAll('.nav-item').forEach(el => el.classList.remove('active'));
  document.querySelectorAll('.sidebar-project').forEach(el => el.classList.remove('active'));
  
  if(page === 'dashboard') {
    document.querySelectorAll('.nav-item')[0].classList.add('active');
    document.getElementById('page-title').textContent = 'Dashboard';
    renderDashboard();
  } else if(page === 'my-tasks') {
    document.querySelectorAll('.nav-item')[1].classList.add('active');
    document.getElementById('page-title').textContent = 'My Tasks';
    renderMyTasks();
  } else if(page === 'members') {
    document.querySelectorAll('.nav-item')[2].classList.add('active');
    document.getElementById('page-title').textContent = 'Team Members';
    renderMembers();
  } else if(page === 'project') {
    const proj = state.projects.find(p => p.id === projectId);
    document.getElementById('page-title').textContent = proj.name;
    document.getElementById(`proj-nav-${projectId}`)?.classList.add('active');
    renderProject(projectId);
  }
  
  renderSidebar();
  updateMyTasksCount();
}

function renderSidebar() {
  const container = document.getElementById('sidebar-projects');
  container.innerHTML = state.projects.map(p => `
    <div class="sidebar-project${state.currentProject === p.id && state.currentPage === 'project' ? ' active' : ''}" 
         id="proj-nav-${p.id}" onclick="navigate('project',${p.id})">
      <span class="proj-dot" style="background:${p.color}"></span>
      <span style="white-space:nowrap;overflow:hidden;text-overflow:ellipsis;font-size:13px">${p.name}</span>
    </div>
  `).join('');
}

function updateMyTasksCount() {
  const mine = state.tasks.filter(t => t.assignee === state.currentUser.id).length;
  document.getElementById('my-tasks-count').textContent = mine;
}

// ========== DRAG & DROP ==========
let draggedTaskId = null;
function dragStart(e, id) { draggedTaskId = id; e.dataTransfer.effectAllowed='move'; }
function dropTask(e, col, projectId) {
  e.preventDefault();
  if(!draggedTaskId) return;
  const task = state.tasks.find(t => t.id === draggedTaskId);
  if(task && task.projectId === projectId) {
    task.column = col;
    toast(`Moved to "${col}"`, 'success');
    renderProject(projectId);
  }
  draggedTaskId = null;
}

        ${t.label ? `<div>
          <div class="task-detail-label">Label</div>
          <span class="badge" style="background:var(--accent-bg);color:var(--accent2)">${t.label}</span>

      </div>

      <div class="task-detail-label">Comments (${(t.comments||[]).length})</div>
      ${(t.comments||[]).length === 0 ? `<p style="font-size:13px;color:var(--muted)">No comments yet.</p>` : ''}
      ${(t.comments||[]).map(c => `
        <div class="comment-box">
          <div style="display:flex;justify-content:space-between">
            <span class="comment-author">${c.author}</span>

        </div>
      `).join('')}

      </div>
    </div>

}

// ========== ACTIONS ==========
function createProject() {
  const name = document.getElementById('m-proj-name')?.value.trim();
  const desc = document.getElementById('m-proj-desc')?.value.trim();
  const color = document.getElementById('m-proj-color')?.value;
  if(!name) { toast('Project name is required', 'error'); return; }
  const id = Math.max(0, ...state.projects.map(p=>p.id)) + 1;
  state.projects.push({id, name, desc: desc||'', color, members:[state.currentUser.id], columns:['To Do','In Progress','Review','Done']});
  closeModal();
  renderSidebar();
  navigate('project', id);
  toast(`Project "${name}" created`, 'success');
}

function updateProject(id) {
  const proj = state.projects.find(p => p.id === id);
  proj.name = document.getElementById('m-proj-name')?.value.trim() || proj.name;
  proj.desc = document.getElementById('m-proj-desc')?.value.trim() || proj.desc;
  closeModal();
  renderSidebar();
  navigate('dashboard');
  toast('Project updated', 'success');
}

function deleteProject(id) {
  if(!confirm('Delete this project and all its tasks?')) return;
  state.projects = state.projects.filter(p => p.id !== id);
  state.tasks = state.tasks.filter(t => t.projectId !== id);
  closeModal();
  renderSidebar();
  navigate('dashboard');
  toast('Project deleted', 'info');
}

function createTask() {
  const title = document.getElementById('m-task-title')?.value.trim();
  if(!title) { toast('Task title is required', 'error'); return; }
  const projectId = parseInt(document.getElementById('m-task-project')?.value);
  const assignee = parseInt(document.getElementById('m-task-assignee')?.value);
  const priority = document.getElementById('m-task-priority')?.value;
  const dueDate = document.getElementById('m-task-due')?.value;
  const label = document.getElementById('m-task-label')?.value;
  const column = document.getElementById('m-task-column')?.value || 'To Do';
  const desc = document.getElementById('m-task-desc')?.value.trim();
  const id = Math.max(0, ...state.tasks.map(t=>t.id)) + 1;
  state.tasks.push({id, projectId, title, desc, column, priority, assignee, dueDate, label, comments:[]});
  closeModal();
  if(state.currentPage === 'project') renderProject(projectId);
  else if(state.currentPage === 'dashboard') renderDashboard();
  else renderMyTasks();
  updateMyTasksCount();
  toast(`Task "${title}" created`, 'success');
}

function deleteTask(id) {
  state.tasks = state.tasks.filter(t => t.id !== id);
  if(state.currentPage === 'project') renderProject(state.currentProject);
  else renderMyTasks();
  updateMyTasksCount();
  toast('Task deleted', 'info');
}

function changeStatus(taskId, col) {
  const t = state.tasks.find(x => x.id === taskId);
  if(t) { t.column = col; toast(`Moved to "${col}"`, 'success'); }
}

function addComment(taskId) {
  const input = document.getElementById('comment-input');
  const text = input?.value.trim();
  if(!text) return;
  const t = state.tasks.find(x => x.id === taskId);
  if(t) {
    if(!t.comments) t.comments = [];
    t.comments.push({author: state.currentUser.name, text, time: 'just now'});
    input.value = '';
    // Re-render comments section
    const container = input.closest('.task-detail-section');
    if(container) openTaskDetail(taskId);
    toast('Comment added', 'success');
  }
}

function addMember() {
  const name = document.getElementById('m-mem-name')?.value.trim();
  const email = document.getElementById('m-mem-email')?.value.trim();
  const role = document.getElementById('m-mem-role')?.value;
  if(!name || !email) { toast('Name and email are required', 'error'); return; }
  const initials = name.split(' ').map(w=>w[0]).join('').toUpperCase().slice(0,2);
  const id = Math.max(0, ...state.members.map(m=>m.id)) + 1;
  const color = AVATARBG[id % AVATARBG.length];
  state.members.push({id, name, email, role, avatar: initials, color});
  closeModal();
  renderMembers();
  toast(`${name} added to team`, 'success');
}

function inviteMember(memberId, projectId) {
  const proj = state.projects.find(p => p.id === projectId);
  if(!proj.members.includes(memberId)) proj.members.push(memberId);
  const member = state.members.find(m => m.id === memberId);
  closeModal();
  toast(`${member.name} invited to project`, 'success');
}

function toggleRole(memberId) {
  const m = state.members.find(x => x.id === memberId);
  if(m) {
    m.role = m.role === 'admin' ? 'member' : 'admin';
    renderMembers();
    toast(`${m.name} is now ${m.role}`, 'info');
  }
}

function addColumn(projectId) {
  const name = prompt('Column name:');
  if(!name) return;
  const proj = state.projects.find(p => p.id === projectId);
  if(proj && !proj.columns.includes(name)) {
    proj.columns.push(name);
    renderProject(projectId);
    toast(`Column "${name}" added`, 'success');
  }
}

function selectColor(el, color) {
  document.querySelectorAll('.color-opt').forEach(o => o.classList.remove('selected'));
  el.classList.add('selected');
  document.getElementById('m-proj-color').value = color;
}

// ========== SEARCH ==========
function handleSearch(q) {
  if(!q.trim()) { navigate(state.currentPage, state.currentProject); return; }
  const results = state.tasks.filter(t => t.title.toLowerCase().includes(q.toLowerCase()) || (t.desc||'').toLowerCase().includes(q.toLowerCase()));
  setContent(`
    <div style="max-width:720px">
      <div class="section-header"><div class="section-title">Search Results (${results.length})</div></div>
      ${results.length === 0 ? '<div class="empty-state"><div class="empty-icon">🔍</div><div class="empty-title">No results</div><div class="empty-desc">Try a different search term.</div></div>' : ''}
      ${results.map(t => renderTaskRow(t)).join('')}
    </div>
  `);
}

// ========== HELPERS ==========
function priorityBadge(p) {
  const map = {high:['p-high','HIGH'],med:['p-med','MED'],low:['p-low','LOW']};
  const [cls, label] = map[p] || ['p-low','LOW'];
  return `<span class="priority-badge ${cls}">${label}</span>`;
}

function statusBadge(col) {
  const colors = {'Done':'var(--green)','In Progress':'var(--amber)','Review':'var(--blue)','To Do':'var(--muted)','Backlog':'var(--muted)'};
  const bg = {
    'Done':'var(--green-bg)','In Progress':'var(--amber-bg)','Review':'var(--blue-bg)',
    'To Do':'var(--bg4)','Backlog':'var(--bg4)'
  };
  const c = colors[col] || 'var(--muted)';
  const b = bg[col] || 'var(--bg4)';
  return `<span style="font-size:11px;font-weight:600;padding:2px 8px;border-radius:4px;background:${b};color:${c}">${col}</span>`;
}

function colColor(col) {
  const map = {'To Do':'#8888a0','In Progress':'#fbbf24','Review':'#60a5fa','Done':'#34d399','Backlog':'#666680'};
  return map[col] || '#7c6af7';
}

function pClass(p) { return {high:'p-high',med:'p-med',low:'p-low'}[p] || 'p-low'; }

function slugify(s) { return s.toLowerCase().replace(/\s+/g,'-'); }

function formatDate(d) {
  try { return new Date(d).toLocaleDateString('en-US',{month:'short',day:'numeric'}); }
  catch { return d; }
}

function setContent(html) {
  const el = document.getElementById('main-content');
  el.innerHTML = html;
  el.classList.remove('fade-in');
  void el.offsetWidth;
  el.classList.add('fade-in');
}

function setTopbarActions(html) {
  document.getElementById('topbar-actions').innerHTML = html;
}

function showModal(title, body, wide=false) {
  document.getElementById('modal-root').innerHTML = `
    <div class="modal-overlay" onclick="if(event.target===this)closeModal()">
      <div class="modal${wide?' ':' '}fade-in" style="max-width:${wide?580:480}px">
        <div class="modal-header">
          <div class="modal-title">${title}</div>
          <button class="modal-close" onclick="closeModal()">✕</button>
        </div>
        <div class="modal-body">${body}</div>
      </div>
    </div>
  `;
}

function closeModal() {
  document.getElementById('modal-root').innerHTML = '';
}

function toast(msg, type='info') {
  const container = document.getElementById('toasts');
  const el = document.createElement('div');
  el.className = `toast toast-${type}`;
  el.innerHTML = `<span>${type==='success'?'✓':type==='error'?'✕':'ℹ'}</span> ${msg}`;
  container.appendChild(el);
  setTimeout(() => el.remove(), 3000);
}
